# On-device Python dependencies (run ON the MaixCAM, not the PC).
#
# Install once per device (survives reboot; lives on the SD card):
#     pip3 install -r /root/aural-ai/requirements_device.txt
# or individually:
#     pip3 install gtts
#
# WHY THIS MATTERS: the device is audio-only (screen removed). Without gTTS,
# every DYNAMIC phrase — scene descriptions, QRIS read-outs, the spoken Wi-Fi
# URL, the "AuralAI siap" cue — silently falls back to nothing. Only the 118
# pre-recorded object chimes in /root/audio/*.wav would play. Installing gTTS
# lights up the whole speech pipeline (utils/tts.py → AudioManager).

# Cloud TTS (Indonesian). Needs internet at synthesis time; results are cached
# under /root/audio/tts_cache so a given phrase is only synthesized once.
gtts>=2.5

# mDNS / .local publishing for http://<device_name>.local:8080.
# NOTE: already present on the stock MaixCAM image (zeroconf 0.39.4) — listed
# here for completeness / reflash reproducibility.
zeroconf>=0.39
